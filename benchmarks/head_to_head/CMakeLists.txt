cmake_minimum_required(VERSION 3.28)
project(anndata_head_to_head LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndata 0.1 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(head_to_head cpp_driver.cpp)
target_compile_features(head_to_head PRIVATE cxx_std_20)
target_compile_options(head_to_head PRIVATE -ffp-contract=off)
target_link_libraries(head_to_head PRIVATE anndata::anndata OpenSSL::Crypto)
