cmake_minimum_required(VERSION 3.28)
project(streaming_probe LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ANNDATA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ANNDATA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ANNDATA_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
if(NOT ANNDATA_SOURCE)
  message(FATAL_ERROR "Set ANNDATA_SOURCE to the source snapshot being measured")
endif()
add_subdirectory("${ANNDATA_SOURCE}" anndata-build)
add_executable(streaming_probe ../streaming.cpp link_shim.cpp)
target_compile_definitions(streaming_probe PRIVATE ANNDATA_STREAM_CANDIDATE=$<BOOL:${ANNDATA_STREAM_CANDIDATE}>)
target_link_libraries(streaming_probe PRIVATE anndata::anndata ${CMAKE_DL_LIBS})
find_package(Threads REQUIRED)
add_executable(streaming_parallel ../streaming_parallel.cpp link_shim.cpp)
target_compile_definitions(streaming_parallel PRIVATE ANNDATA_STREAM_CANDIDATE=$<BOOL:${ANNDATA_STREAM_CANDIDATE}>)
target_link_libraries(streaming_parallel PRIVATE anndata::anndata Threads::Threads ${CMAKE_DL_LIBS})
