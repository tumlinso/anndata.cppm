cmake_minimum_required(VERSION 3.28)
project(streaming_probe LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ANNDATAPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ANNDATAPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ANNDATAPP_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
if(NOT ANNDATAPP_SOURCE)
  message(FATAL_ERROR "Set ANNDATAPP_SOURCE to the source snapshot being measured")
endif()
add_subdirectory("${ANNDATAPP_SOURCE}" anndatapp-build)
add_executable(streaming_probe ../streaming.cpp link_shim.cpp)
target_compile_definitions(streaming_probe PRIVATE ANNDATAPP_STREAM_CANDIDATE=$<BOOL:${ANNDATAPP_STREAM_CANDIDATE}>)
target_link_libraries(streaming_probe PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
find_package(Threads REQUIRED)
add_executable(streaming_parallel ../streaming_parallel.cpp link_shim.cpp)
target_compile_definitions(streaming_parallel PRIVATE ANNDATAPP_STREAM_CANDIDATE=$<BOOL:${ANNDATAPP_STREAM_CANDIDATE}>)
target_link_libraries(streaming_parallel PRIVATE anndatapp::anndatapp Threads::Threads ${CMAKE_DL_LIBS})
