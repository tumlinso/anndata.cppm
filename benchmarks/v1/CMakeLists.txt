cmake_minimum_required(VERSION 3.28)
project(anndatapp_v1_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ANNDATAPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ANNDATAPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ANNDATAPP_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
if(NOT ANNDATAPP_SOURCE)
  message(FATAL_ERROR "Set ANNDATAPP_SOURCE to the immutable source snapshot")
endif()
add_subdirectory("${ANNDATAPP_SOURCE}" anndatapp-build)
find_package(Threads REQUIRED)
add_executable(v1_benchmark main.cpp core_cases.cpp storage_cases.cpp)
target_compile_features(v1_benchmark PRIVATE cxx_std_20)
target_compile_definitions(v1_benchmark PRIVATE V1_NEW_CORE=$<BOOL:${V1_NEW_CORE}>)
target_link_libraries(v1_benchmark PRIVATE anndatapp::anndatapp Threads::Threads ${CMAKE_DL_LIBS})
add_library(v1_io_counters SHARED io_counters.c)
target_link_libraries(v1_io_counters PRIVATE ${CMAKE_DL_LIBS})
add_library(v1_alloc_counters SHARED alloc_counters.cpp)
target_compile_features(v1_alloc_counters PRIVATE cxx_std_20)
find_package(PkgConfig REQUIRED)
pkg_check_modules(V1_OBSERVER_DEPS REQUIRED IMPORTED_TARGET hdf5 libzstd zlib)
if(APPLE)
  target_link_libraries(v1_io_counters PRIVATE PkgConfig::V1_OBSERVER_DEPS)
endif()
add_executable(v1_observer_check observer_check.cpp)
target_compile_features(v1_observer_check PRIVATE cxx_std_20)
target_link_libraries(v1_observer_check PRIVATE PkgConfig::V1_OBSERVER_DEPS ${CMAKE_DL_LIBS})
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/observer-paths.json" CONTENT
  "{\"interposer\":\"$<TARGET_FILE:v1_io_counters>\",\"alloc_interposer\":\"$<TARGET_FILE:v1_alloc_counters>\"}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/toolchain-paths.json" CONTENT
  "{\"compiler\":\"${CMAKE_CXX_COMPILER}\",\"scanner\":\"${CMAKE_CXX_COMPILER_CLANG_SCAN_DEPS}\"}\n")
configure_file(core_consumer.cpp core_consumer.cpp COPYONLY)
add_executable(v1_core_consumer "${CMAKE_CURRENT_BINARY_DIR}/core_consumer.cpp")
target_compile_features(v1_core_consumer PRIVATE cxx_std_20)
if(TARGET anndatapp::core)
  target_link_libraries(v1_core_consumer PRIVATE anndatapp::core)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumer-linkage.txt" "anndatapp::core")
else()
  target_link_libraries(v1_core_consumer PRIVATE anndatapp::anndatapp)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/consumer-linkage.txt" "anndatapp::anndatapp")
endif()
