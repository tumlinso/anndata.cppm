cmake_minimum_required(VERSION 3.28)
project(completeness_probe LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ANNDATAPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ANNDATAPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ANNDATAPP_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
if(NOT ANNDATAPP_SOURCE)
  message(FATAL_ERROR "Set ANNDATAPP_SOURCE to the snapshot being measured")
endif()
add_subdirectory("${ANNDATAPP_SOURCE}" anndatapp-build)
find_package(Threads REQUIRED)
add_executable(completeness_probe ../completeness.cpp)
target_link_libraries(completeness_probe PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
add_executable(completeness_adaptive ../completeness_adaptive.cpp)
target_link_libraries(completeness_adaptive PRIVATE anndatapp::anndatapp Threads::Threads ${CMAKE_DL_LIBS})
