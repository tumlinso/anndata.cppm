cmake_minimum_required(VERSION 3.28)
project(anndata_consumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndata 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE anndata::anndata)
add_executable(consumer_records records.cpp)
target_link_libraries(consumer_records PRIVATE anndata::anndata)
add_executable(consumer_facade facade.cpp)
target_link_libraries(consumer_facade PRIVATE anndata::anndata)
add_executable(consumer_backed backed.cpp)
target_link_libraries(consumer_backed PRIVATE anndata::anndata)
if(ANNDATA_HAS_HDF5)
  add_executable(consumer_h5ad h5ad.cpp)
  target_link_libraries(consumer_h5ad PRIVATE anndata::anndata)
endif()
if(ANNDATA_HAS_ZARR)
  add_executable(consumer_zarr zarr.cpp)
  target_link_libraries(consumer_zarr PRIVATE anndata::anndata)
endif()
