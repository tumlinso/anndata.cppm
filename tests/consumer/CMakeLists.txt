cmake_minimum_required(VERSION 3.28)
project(anndatapp_consumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndatapp 1.0 CONFIG REQUIRED COMPONENTS core)
find_package(anndatapp 1.0 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE anndatapp::core)
add_executable(consumer_records records.cpp)
target_link_libraries(consumer_records PRIVATE anndatapp::core)
add_executable(consumer_facade facade.cpp)
target_link_libraries(consumer_facade PRIVATE anndatapp::anndatapp)
add_executable(consumer_backed backed.cpp)
target_link_libraries(consumer_backed PRIVATE anndatapp::anndatapp)
add_executable(consumer_streaming streaming.cpp)
target_link_libraries(consumer_streaming PRIVATE anndatapp::anndatapp)
if(ANNDATAPP_HAS_HDF5)
  add_executable(consumer_h5ad h5ad.cpp)
  target_link_libraries(consumer_h5ad PRIVATE anndatapp::anndatapp)
endif()
if(ANNDATAPP_HAS_ZARR)
  add_executable(consumer_zarr zarr.cpp)
  target_link_libraries(consumer_zarr PRIVATE anndatapp::anndatapp)
endif()
add_library(facade_bridge INTERFACE)
target_link_libraries(facade_bridge INTERFACE anndatapp::anndatapp)
install(TARGETS facade_bridge EXPORT facade_bridge_targets)
install(EXPORT facade_bridge_targets NAMESPACE bridge:: DESTINATION lib/cmake/bridge)
